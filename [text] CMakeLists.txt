[cmake]
cmake_minimum_required(VERSION 3.22)

project(ModelCycles6CH VERSION 1.0.0)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Scarica automaticamente JUCE 9.0.5 durante la configurazione
include(FetchContent)
FetchContent_Declare(
    JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG        9.0.5
    GIT_SHALLOW    TRUE
)
FetchContent_MakeAvailable(JUCE)

# Formati supportati: AU + VST3 su macOS, VST3 su Windows/Linux
if(APPLE)
    set(PLUGIN_FORMATS AU VST3)
else()
    set(PLUGIN_FORMATS VST3)
endif()

juce_add_plugin(ModelCycles6CH
    COMPANY_NAME               "ModelCyclesCommunity"
    IS_SYNTH                   FALSE
    NEEDS_MIDI_INPUT           TRUE
    NEEDS_MIDI_OUTPUT          TRUE
    IS_MIDI_EFFECT             FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    COPY_PLUGIN_AFTER_BUILD    FALSE
    PLUGIN_MANUFACTURER_CODE   Mc6c
    PLUGIN_CODE                Mc6h
    FORMATS                    ${PLUGIN_FORMATS}
    PRODUCT_NAME               "ModelCycles 6CH"
    VST3_CATEGORIES            Fx Tools
    # kAudioUnitType_Effect permette di caricare l'AU direttamente sulle tracce audio in Logic Pro
    AU_MAIN_TYPE               kAudioUnitType_Effect
)

target_sources(ModelCycles6CH
    PRIVATE
        Source/ModelCycles6CH.cpp
)

target_compile_definitions(ModelCycles6CH
    PUBLIC
        JUCE_WEB_BROWSER=0
        JUCE_USE_CURL=0
        JUCE_VST3_CAN_REPLACE_VST2=0
)

target_link_libraries(ModelCycles6CH
    PRIVATE
        juce::juce_audio_processors
        juce::juce_audio_devices
        juce::juce_gui_basics
        juce::juce_audio_basics
        juce::juce_core
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags
)
